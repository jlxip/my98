cmake_minimum_required(VERSION 3.20)
project(my98_libtorrent LANGUAGES CXX)
find_package(Python3 COMPONENTS Interpreter Development REQUIRED)
set(boost-python-module-name "python${Python3_VERSION_MAJOR}${Python3_VERSION_MINOR}" CACHE STRING "Boost.Python component")
find_package(Boost REQUIRED COMPONENTS ${boost-python-module-name})
find_package(LibtorrentRasterbar CONFIG REQUIRED)
Python3_add_library(my98_libtorrent MODULE WITH_SOABI signed_record.cpp)
target_compile_features(my98_libtorrent PRIVATE cxx_std_17)
target_link_libraries(my98_libtorrent PRIVATE LibtorrentRasterbar::torrent-rasterbar Boost::${boost-python-module-name})
install(TARGETS my98_libtorrent DESTINATION lib/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages)
